npm 命令:
  npm run test:e2e              # 運行所有 E2E 測試
  npm run test:e2e:ui          # UI 模式（推薦開發用）
  npm run test:e2e:debug       # 調試模式
  npm run test:e2e:report      # 查看 HTML 報告

腳本命令:
  ./test.sh run                # 運行所有測試
  ./test.sh ui                 # UI 模式
  ./test.sh debug              # 調試模式
  ./test.sh report             # 查看報告
  ./test.sh chromium           # 僅 Chromium
  ./test.sh firefox            # 僅 Firefox
  ./test.sh webkit             # 僅 WebKit
  ./test.sh match "測試名稱"   # 搜索特定測試
